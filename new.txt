CREATE OR REPLACE FUNCTION public.handle_parent_challenge_completion()
RETURNS TRIGGER
LANGUAGE plpgsql
AS $$
DECLARE
  v_parent_challenge RECORD;
  v_total_children INT;
  v_completed_children INT;
  v_already_completed_parent BOOLEAN;
  v_parent_reward INT := 100; -- <-- ¡Recompensa final por el desafío! (ajústala)
BEGIN
  -- 1. Solo nos importa si el reto fue 'completado'
  IF NEW.resultado <> 'completado' THEN
    RETURN NEW;
  END IF;

  -- 2. Buscar el desafío "padre" (tipo 5) al que pertenece este "hijo" (NEW.id_reto)
  --    Usa la misma lógica de fechas que tu 'desafio_provider'
  SELECT 
    padre.id_reto, 
    padre.fecha_inicio, 
    padre.fecha_final
    -- (Puedes añadir una columna 'recompensa_final' a tu tabla 'reto' y leerla aquí)
  INTO v_parent_challenge
  FROM public.reto AS hijo
  JOIN public.reto AS padre ON 
    hijo.fecha_inicio >= padre.fecha_inicio AND 
    hijo.fecha_final <= padre.fecha_final AND
    padre.tipo_reto = 5 AND -- Es un padre
    padre.especial = true AND
    padre.activo = true
  WHERE 
    hijo.id_reto = NEW.id_reto AND
    hijo.tipo_reto <> 5; -- Asegurarse de que el reto completado es un "hijo"

  -- 3. Si no se encontró un padre (es un reto normal del home), salir.
  IF v_parent_challenge IS NULL THEN
    RETURN NEW;
  END IF;

  -- 4. Verificar si el usuario YA completó este reto padre (evita recompensas duplicadas)
  SELECT EXISTS (
    SELECT 1
    FROM public.intento_reto
    WHERE id_usuario = NEW.id_usuario
      AND id_reto = v_parent_challenge.id_reto
      AND resultado = 'completado'
  ) INTO v_already_completed_parent;

  IF v_already_completed_parent THEN
    RETURN NEW;
  END IF;

  -- 5. Contar el TOTAL de retos "hijos" para este padre
  SELECT COUNT(*)
  INTO v_total_children
  FROM public.reto
  WHERE 
    tipo_reto <> 5 AND
    especial = false AND
    activo = true AND
    fecha_inicio >= v_parent_challenge.fecha_inicio AND
    fecha_final <= v_parent_challenge.fecha_final;

  -- 6. Contar cuántos de esos "hijos" ha completado el usuario
  SELECT COUNT(DISTINCT ir.id_reto)
  INTO v_completed_children
  FROM public.intento_reto ir
  JOIN public.reto r ON ir.id_reto = r.id_reto
  WHERE 
    ir.id_usuario = NEW.id_usuario AND
    ir.resultado = 'completado' AND
    r.tipo_reto <> 5 AND
    r.especial = false AND
    r.activo = true AND
    r.fecha_inicio >= v_parent_challenge.fecha_inicio AND
    r.fecha_final <= v_parent_challenge.fecha_final;

  -- 7. ¡LA GRAN VERIFICACIÓN!
  --    Si el total de hijos es > 0 y el usuario los ha completado TODOS...
  IF v_total_children > 0 AND v_completed_children = v_total_children THEN
    
    -- ¡Otorga la recompensa!
    -- Insertamos un nuevo intento para el reto PADRE.
    -- El trigger 'handle_new_intento' (modificado) lo verá,
    -- entrará en la sección 'ELSE' y respetará la 'experiencia_obtenida'.
    INSERT INTO public.intento_reto (
      id_usuario,
      id_reto,
      resultado,
      experiencia_obtenida -- ¡Aquí va la recompensa!
    )
    VALUES (
      NEW.id_usuario,
      v_parent_challenge.id_reto,
      'completado',
      v_parent_reward 
    );
    
  END IF;

  RETURN NEW;
END;
$$;

CREATE TRIGGER on_intento_complete_parent
